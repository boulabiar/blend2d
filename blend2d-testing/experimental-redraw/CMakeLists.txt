# Review-only fixtures. Their private hooks are available through static linking.
if(NOT BLEND2D_STATIC)
  message(FATAL_ERROR "Experimental redraw fixtures require BLEND2D_STATIC=ON")
endif()
if(NOT UNIX)
  message(FATAL_ERROR "Experimental redraw fixtures currently require a POSIX host")
endif()
find_package(Threads REQUIRED)
foreach(pair IN ITEMS "equivalence:replay-equivalence.cpp" "split_fill:split-fill-equivalence.cpp" "render:render-equivalence.cpp" "type_predicates:type-predicate-table.cpp")
  string(REPLACE ":" ";" parts "${pair}")
  list(GET parts 0 suffix)
  list(GET parts 1 source)
  add_executable(bl_review_${suffix} "${CMAKE_CURRENT_LIST_DIR}/${source}")
  target_compile_features(bl_review_${suffix} PRIVATE cxx_std_20)
  target_compile_definitions(bl_review_${suffix} PRIVATE BL_STATIC)
  target_compile_options(bl_review_${suffix} PRIVATE ${BLEND2D_SANITIZE_CFLAGS})
  target_link_options(bl_review_${suffix} PRIVATE ${BLEND2D_SANITIZE_LFLAGS})
  target_link_libraries(bl_review_${suffix} PRIVATE blend2d::blend2d Threads::Threads ${CMAKE_DL_LIBS})
  add_test(NAME bl_review_${suffix} COMMAND bl_review_${suffix})
endforeach()
